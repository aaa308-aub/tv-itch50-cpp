// A mock run of the parser is done at every start of this process to fill the
// page cache first. We want to benchmark the parser, not the disk I/O.
static void warmCache() {
	static bool is_already_warmed = false;
	if (is_already_warmed) {
		return;
	} else {
		is_already_warmed = true;
	}

	std::string path = BENCHMARK_FILEPATH;
	benchmark::DoNotOptimize(path);

	HandlerAllUndef h;
	itch::Parser p(path, h);
	std::uint64_t i = 0;
	while (p.next()) {
		++i;
	}
	benchmark::DoNotOptimize(i);
}

static void benchmarkAllUndef(benchmark::State& state) {
	warmCache();

	std::string path = BENCHMARK_FILEPATH;
	benchmark::DoNotOptimize(path);

	for (auto _ : state) { // This runs a timer.
    	HandlerAllUndef h;
     	itch::Parser p(path, h);
      	std::uint64_t i = 0; // May be redundant.

      	while (p.next()) {
        	p.callHandler();
         	++i;
        }

       	benchmark::DoNotOptimize(i);
        benchmark::ClobberMemory();
   }
}

static void benchmarkAllEmpty(benchmark::State& state) {
	warmCache();

	std::string path = BENCHMARK_FILEPATH;
	benchmark::DoNotOptimize(path);

	for (auto _ : state) {
    	HandlerAllEmpty h;
     	itch::Parser p(path, h);

      	while (p.next()) {
        	p.callHandler();
        }

        benchmark::ClobberMemory();
   }
}

static void benchmarkAllCopy(benchmark::State& state) {

	benchmark::DoNotOptimize(path);

	for (auto _ : state) {
    	HandlerAllCopy h;
     	itch::Parser p(path, h);

      	while (p.next()) {
        	p.callHandler();
        }

        benchmark::ClobberMemory();
   }
}

BENCHMARK(benchmarkAllUndef);
BENCHMARK(benchmarkAllEmpty);
BENCHMARK(benchmarkAllCopy);
BENCHMARK_MAIN();
